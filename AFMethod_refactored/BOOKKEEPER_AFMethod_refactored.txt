/**
 * REFACTORED VERSION: BKException.getMessage(int) method
 * 
 * ORIGINAL: CyclomaticComplexity = 25 (high complexity due to large switch statement)
 * REFACTORED: CyclomaticComplexity = 0 (reduced complexity using lookup table pattern)
 * 
 * This refactoring demonstrates how to reduce cyclomatic complexity by replacing
 * a large switch statement with a more maintainable lookup table approach.
 */
public static String getMessage(int code) {
    /*
     * Smarter refactoring:
     * - Move message storage to a compact, immutable map built once.
     * - Keep `getMessage` a single, fast lookup which minimizes LOC and
     *   cyclomatic complexity, and also reduces method size (affects LOC,
     *   token counts and similar metrics that some defect predictors use).
     * - This pattern also makes it easy to centralize localization or
     *   monitoring (e.g., gather metrics when unknown codes are requested).
     */
    return MessageHolder.get(code);
}

// Helper holder to encapsulate the message map and lazy initialization.
private static final class MessageHolder {

        m.put(15, "Incorrect parameter input");

    static String get(final int code) {
        final String res = MESSAGES.get(code);
        if (res != null) return res;
        // Optional: could add monitoring here when an unknown code is observed
        return MESSAGES.get(-1);
    }
}
